160 void startBinding(unsigned powerPin, unsigned rxPin) {
161 // switch off serial interface
162 if (serial) {
163 log("startBinding");
164
165 pinMode(rxPin, OUTPUT); // sets the digital pin as output
166 digitalWrite(rxPin, LOW); // make sure that the pin off
167 pinMode(powerPin, OUTPUT); // sets the digital pin as output
168 digitalWrite(powerPin, LOW); // make sure that the pin off
169 digitalWrite(rxPin, HIGH); // set initial state to high
170 delay(2000);
171
172 // To put a receiver into bind mode, within 200ms of power application
173 // the host device needs to issue a series of falling pulses
174 pinMode(powerPin, OUTPUT); // sets the digital pin as output
175 digitalWrite(powerPin, HIGH); // make sure that the pin off
176 delay(50);
177
178 log("-> number of pulses: ", bindMode);
179
180 for (int j = 0; j < bindMode; j++) {
181 digitalWrite(rxPin, HIGH); // sets the digital pin on
182 delayMicroseconds(BINDING_PULSE_DELAY_MS); // waits
183 digitalWrite(rxPin, LOW); // sets the digital pin off
184 delayMicroseconds(BINDING_PULSE_DELAY_MS); // waits
185 }
186
187 log("-> number of pulses DONE");
188
189 delay(500);
190 pinMode(rxPin, INPUT); // sets the pin as input
191 }

195 bool getFrame(int transactionTimeMs = DEFAULT_RECEIVING_TIMEOUT) {
196 short inByte;
197 byte inData[SEND_BUFFER_SIZE];
198 bool result = false;
199 long available = serial->available();
200
201 // 16-byte data packet every 11ms or 22ms
202 if (available >= 16) {
203 timeOfLastRead = millis();
204 // resynchronize and use last data
205 if (!processAllData && available > 16) {
206 long diff = available - 16;
207 log("skipping number of bytes:", diff);
208 // skip unnecessary data
209 for (int j = 0; j < diff; j++) serial->read();
210 }
211
212 // read the latest data packet
213 inByte = serial->readBytes(inData, 16);
214 if (inByte != 16) {
215 log("We could not read all data");
216 result = false;
217 } else {
218 // check if we processed the data within indicated time period
219 result = isConnected(transactionTimeMs);
220 if (result) {
221 parseFrame(inData);
222 // check if the frame is valid
223 result = isValidSystem(this->system);
224 status = Receiving;
225
226 // log status
227 logFrame(available, result);
228 } else {
229 log("Frame ignored because of timeout");
230 }
231 }
232 }
233
234 return result;
235 }

424 bool parseFrame(Data* inData) {
425 Data* data = (Data*)inData;
426 // a frame is 16 bytes -> 7 channels + fades
427 if (isInternal()) {
428 this->fades = data->header.internal.fades;
429 System recevedSystem = (System)data->header.internal.system;
430 if (!isSystemReported || recevedSystem != getSystem()) {
431 logHex("System from the Satellite:", recevedSystem);
432 isSystemReported = true;
433 }
434 if (recevedSystem != getSystem()) {
435 if (isValidSystem(recevedSystem))
436 setSystem(recevedSystem);
437 else
438 logHex("Unexpected system", recevedSystem);
439 }
440 } else {
441 this->fades = data->header.fades;
442 }
443
444 uint16_t channelShift = is2048() ? 11 : 10;
445 for (int i = 0; i < 7; i++) {
446 uint16_t inValue = data->values[i];
447 swapBytes(&inValue);
448 uint16_t channelID = (inValue & maskCHANID) >> channelShift;
449 uint16_t channelValue = inValue & maskVALUE;
450
451 if (channelID >= 0 && channelID < MAX_CHANNELS) {
452 channelValues[channelID] = channelValue;
453 }
454 }
455 return true;
456 }
458 Data* getSendBuffer(boolean auxData) {
459 // Clear only the values array and header
460 for (int i = 0; i < 7; ++i) dataPacket.values[i] = 0;
461 dataPacket.header.fades = 0;
462
463 uint16_t channelShift = is2048() ? 11 : 10;
464 if (auxData) {
465 for (int j = 6; j < MAX_CHANNELS && (j - 6) < 7; ++j) {
466 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
467 swapBytes(&value);
468 dataPacket.values[j - 6] = value;
469 }
470 } else {
471 for (int j = 0; j < 7; ++j) {
472 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
473 swapBytes(&value);
474 dataPacket.values[j] = value;
475 }
476 }
477
478 if (isInternal()) {
479 dataPacket.header.internal.fades = this->fades;
480 dataPacket.header.internal.system = this->system;
481 } else {
482 dataPacket.header.fades = this->fades;
483 }
484
485 return &dataPacket;
486 }
