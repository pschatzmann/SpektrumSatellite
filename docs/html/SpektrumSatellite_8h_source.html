181 void startBinding(unsigned powerPin, unsigned rxPin) {
182 // switch off serial interface
183 if (serial) {
184 log("startBinding");
185
186 pinMode(rxPin, OUTPUT); // sets the digital pin as output
187 digitalWrite(rxPin, LOW); // make sure that the pin off
188 pinMode(powerPin, OUTPUT); // sets the digital pin as output
189 digitalWrite(powerPin, LOW); // make sure that the pin off
190 digitalWrite(rxPin, HIGH); // set initial state to high
191 delay(2000);
192
193 // To put a receiver into bind mode, within 200ms of power application
194 // the host device needs to issue a series of falling pulses
195 pinMode(powerPin, OUTPUT); // sets the digital pin as output
196 digitalWrite(powerPin, HIGH); // make sure that the pin off
197 delay(50);
198
199 log("-> number of pulses: ", bindMode);
200
201 for (int j = 0; j < bindMode; j++) {
202 digitalWrite(rxPin, HIGH); // sets the digital pin on
203 delayMicroseconds(BINDING_PULSE_DELAY_MS); // waits
204 digitalWrite(rxPin, LOW); // sets the digital pin off
205 delayMicroseconds(BINDING_PULSE_DELAY_MS); // waits
206 }
207
208 log("-> number of pulses DONE");
209
210 delay(500);
211 pinMode(rxPin, INPUT); // sets the pin as input
212 }

216 bool getFrame(int transactionTimeMs = DEFAULT_RECEIVING_TIMEOUT) {
217 short inByte;
218 byte inData[SEND_BUFFER_SIZE];
219 bool result = false;
220 long available = serial->available();
221
222 // 16-byte data packet every 11ms or 22ms
223 if (available >= 16) {
224 timeOfLastRead = millis();
225 // resynchronize and use last data
226 if (!processAllData && available > 16) {
227 long diff = available - 16;
228 log("skipping number of bytes:", diff);
229 // skip unnecessary data
230 for (int j = 0; j < diff; j++) serial->read();
231 }
232
233 // read the latest data packet
234 inByte = serial->readBytes(inData, 16);
235 if (inByte != 16) {
236 log("We could not read all data");
237 result = false;
238 } else {
239 // check if we processed the data within indicated time period
240 result = isConnected(transactionTimeMs);
241 if (result) {
242 parseFrame(inData);
243 // check if the frame is valid
244 result = isValidSystem(this->system);
245 status = Receiving;
246
247 // log status
248 logFrame(available, result);
249 } else {
250 log("Frame ignored because of timeout");
251 }
252 }
253 }
254
255 return result;
256 }

445 bool parseFrame(Data* inData) {
446 Data* data = (Data*)inData;
447 // a frame is 16 bytes -> 7 channels + fades
448 if (isInternal()) {
449 this->fades = data->header.internal.fades;
450 System recevedSystem = (System)data->header.internal.system;
451 if (!isSystemReported || recevedSystem != getSystem()) {
452 logHex("System from the Satellite:", recevedSystem);
453 isSystemReported = true;
454 }
455 if (recevedSystem != getSystem()) {
456 if (isValidSystem(recevedSystem))
457 setSystem(recevedSystem);
458 else
459 logHex("Unexpected system", recevedSystem);
460 }
461 } else {
462 this->fades = data->header.fades;
463 }
464
465 uint16_t channelShift = is2048() ? 11 : 10;
466 for (int i = 0; i < 7; i++) {
467 uint16_t inValue = data->values[i];
468 swapBytes(&inValue);
469 uint16_t channelID = (inValue & maskCHANID) >> channelShift;
470 uint16_t channelValue = inValue & maskVALUE;
471
472 if (channelID >= 0 && channelID < MAX_CHANNELS) {
473 channelValues[channelID] = channelValue;
474 }
475 }
476 return true;
477 }
479 Data* getSendBuffer(boolean auxData) {
480 // Clear only the values array and header
481 for (int i = 0; i < 7; ++i) dataPacket.values[i] = 0;
482 dataPacket.header.fades = 0;
483
484 uint16_t channelShift = is2048() ? 11 : 10;
485 if (auxData) {
486 for (int j = 6; j < MAX_CHANNELS && (j - 6) < 7; ++j) {
487 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
488 swapBytes(&value);
489 dataPacket.values[j - 6] = value;
490 }
491 } else {
492 for (int j = 0; j < 7; ++j) {
493 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
494 swapBytes(&value);
495 dataPacket.values[j] = value;
496 }
497 }
498
499 if (isInternal()) {
500 dataPacket.header.internal.fades = this->fades;
501 dataPacket.header.internal.system = this->system;
502 } else {
503 dataPacket.header.fades = this->fades;
504 }
505
506 return &dataPacket;
507 }
