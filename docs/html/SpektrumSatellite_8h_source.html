181 void startBinding(unsigned powerPin, unsigned rxPin) {
182 log("startBinding");
183 status = Binding;
184
185 pinMode(rxPin, OUTPUT); // sets the digital pin as output
186 digitalWrite(rxPin, LOW); // make sure that the pin off
187 pinMode(powerPin, OUTPUT); // sets the digital pin as output
188 digitalWrite(powerPin, LOW); // make sure that the pin off
189 digitalWrite(rxPin, HIGH); // set initial state to high
190 delay(2000);
191
192 // To put a receiver into bind mode, within 200ms of power application
193 // the host device needs to issue a series of falling pulses
194 pinMode(powerPin, OUTPUT); // sets the digital pin as output
195 digitalWrite(powerPin, HIGH); // make sure that the pin off
196 delay(50);
197
198 log("-> number of pulses: ", bindMode);
199
200 for (int j = 0; j < bindMode; j++) {
201 digitalWrite(rxPin, HIGH); // sets the digital pin on
202 delayMicroseconds(BINDING_PULSE_DELAY_US); // waits
203 digitalWrite(rxPin, LOW); // sets the digital pin off
204 delayMicroseconds(BINDING_PULSE_DELAY_US); // waits
205 }
206
207 log("-> number of pulses DONE");
208
209 delay(500);
210 pinMode(rxPin, INPUT); // sets the pin as input
211
212 status = NotConnected;
213 }

218 bool getFrame(int transactionTimeMs = DEFAULT_RECEIVING_TIMEOUT) {
219 const long frameSize = SEND_BUFFER_SIZE;
220 byte inData[SEND_BUFFER_SIZE];
221 long available = serial->available();
222
223 // 16-byte data packet every 11ms or 22ms
224 if (available < frameSize) {
225 if (status == Receiving && !isConnected(transactionTimeMs)) {
226 log("Connection lost");
227 status = NotConnected;
228 }
229 return false;
230 }
231
232 // resynchronize and use last complete frame: we skip only whole frames
233 // so that we keep the frame alignment
234 if (!processAllData && available >= 2 * frameSize) {
235 long diff = (available / frameSize - 1) * frameSize;
236 log("skipping number of bytes:", diff);
237 for (long j = 0; j < diff; j++) serial->read();
238 }
239
240 // read the latest data packet
241 if ((long)serial->readBytes(inData, frameSize) != frameSize) {
242 log("We could not read all data");
243 return false;
244 }
245
246 bool result = parseFrame(inData);
247 if (result) {
248 timeOfLastRead = millis();
249 status = Receiving;
250 } else if (isInternal()) {
251 // invalid system byte: we are most likely not aligned to the frame
252 // start, so we shift by one byte to resynchronize
253 serial->read();
254 }
255
256 // log status
257 logFrame(available, result);
258 return result;
259 }

454 bool parseFrame(Data* inData) {
455 Data* data = (Data*)inData;
456 // a frame is 16 bytes -> 7 channels + fades
457 if (isInternal()) {
458 this->fades = data->header.internal.fades;
459 System recevedSystem = (System)data->header.internal.system;
460 if (!isSystemReported || recevedSystem != getSystem()) {
461 logHex("System from the Satellite:", recevedSystem);
462 isSystemReported = true;
463 }
464 if (recevedSystem != getSystem()) {
465 if (!isValidSystem(recevedSystem)) {
466 logHex("Unexpected system", recevedSystem);
467 return false;
468 }
469 setSystem(recevedSystem);
470 }
471 } else {
472 uint16_t fades = data->header.fades;
473 swapBytes(&fades);
474 this->fades = fades;
475 }
476
477 uint16_t channelShift = is2048() ? 11 : 10;
478 for (int i = 0; i < 7; i++) {
479 uint16_t inValue = data->values[i];
480 swapBytes(&inValue);
481 uint16_t channelID = (inValue & maskCHANID) >> channelShift;
482 uint16_t channelValue = inValue & maskVALUE;
483
484 if (channelID < MAX_CHANNELS) {
485 channelValues[channelID] = channelValue;
486 }
487 }
488 return true;
489 }
491 Data* getSendBuffer(boolean auxData) {
492 // unused slots are 0xFFFF: this decodes to an invalid channel id
493 for (int i = 0; i < 7; ++i) dataPacket.values[i] = 0xFFFF;
494 dataPacket.header.fades = 0;
495
496 uint16_t channelShift = is2048() ? 11 : 10;
497 if (auxData) {
498 for (int j = 6; j < MAX_CHANNELS && (j - 6) < 7; ++j) {
499 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
500 swapBytes(&value);
501 dataPacket.values[j - 6] = value;
502 }
503 } else {
504 for (int j = 0; j < 7; ++j) {
505 uint16_t value = (channelValues[j] & maskVALUE) | (j << channelShift);
506 swapBytes(&value);
507 dataPacket.values[j] = value;
508 }
509 }
510
511 if (isInternal()) {
512 dataPacket.header.internal.fades = this->fades;
513 dataPacket.header.internal.system = this->system;
514 } else {
515 uint16_t fades = this->fades;
516 swapBytes(&fades);
517 dataPacket.header.fades = fades;
518 }
519
520 return &dataPacket;
521 }
